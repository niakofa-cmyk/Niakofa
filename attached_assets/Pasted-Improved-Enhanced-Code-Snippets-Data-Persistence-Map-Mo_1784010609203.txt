// artifacts/pay-it-forward/src/lib/AppContext.tsx — Enhanced persistence
import { createContext, useContext, useState, useEffect, ReactNode } from 'react';

interface AppContextType {
  requests: any[];
  setRequests: (data: any[]) => void;
  // ... other fields
  refreshAllData: () => Promise<void>;
}

const AppContext = createContext<AppContextType | null>(null);

export const AppProvider = ({ children }: { children: ReactNode }) => {
  const [requests, setRequestsState] = useState<any[]>([]);

  const setRequests = (newData: any[]) => {
    setRequestsState(newData);
    localStorage.setItem('niakofa_requests_cache', JSON.stringify(newData)); // Fallback
  };

  const refreshAllData = async () => {
    try {
      const res = await fetch('/api/requests?status=open,claimed,ongoing', {
        headers: authHeaders(),
      });
      if (res.ok) {
        const data = await res.json();
        setRequests(data.requests || data);
      }
    } catch (e) {
      // Load cache on failure
      const cached = localStorage.getItem('niakofa_requests_cache');
      if (cached) setRequests(JSON.parse(cached));
    }
  };

  // Auto-refresh on visibility, focus, mode change
  useEffect(() => {
    refreshAllData();
    const handle = () => refreshAllData();
    document.addEventListener('visibilitychange', handle);
    window.addEventListener('focus', handle);
    return () => {
      document.removeEventListener('visibilitychange', handle);
      window.removeEventListener('focus', handle);
    };
  }, [helperModeActive]); // Re-sync on mode switch

  return (
    <AppContext.Provider value={{ requests, setRequests, refreshAllData /* ... */ }}>
      {children}
    </AppContext.Provider>
  );
};

3. DB / API Hardening (requests.ts example)

// artifacts/api-server/src/routes/requests.ts — Ensure no data loss
router.get('/requests', requireAuth, async (req, res) => {
  const { status = 'open,claimed,ongoing' } = req.query;
  const statuses = (status as string).split(',');

  const data = await db.select().from(requestsTable)
    .where(inArray(requestsTable.status, statuses))
    .orderBy(desc(requestsTable.createdAt));

  res.json({ requests: data, timestamp: Date.now() }); // Add timestamp for client sync
});
