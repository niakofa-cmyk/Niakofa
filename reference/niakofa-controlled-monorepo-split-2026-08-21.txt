Bridge (mock-first, API later)
apps/legacy-rpg/src/integration/niakofa-bridge.ts:
TypeScript

export interface LegacyLaunchContext {
  mode: "mock" | "live";
  familyId?: string;
  characterId?: string;
  sessionToken?: string;
  gameHour?: number;
}

/** Standalone: always mock. Embedded: read query/hash or postMessage from parent. */
export function getLegacyLaunchContext(): LegacyLaunchContext {
  if (typeof window === "undefined") {
    return { mode: "mock", characterId: "kwame-mensah", gameHour: 14 };
  }

  const params = new URLSearchParams(window.location.search);
  const token = params.get("token") ?? undefined;
  const familyId = params.get("familyId") ?? undefined;
  const characterId = params.get("characterId") ?? "kwame-mensah";

  if (token || familyId) {
    return { mode: "live", familyId, characterId, sessionToken: token, gameHour: 14 };
  }

  // Default playable without the main app
  return {
    mode: "mock",
    characterId: "kwame-mensah",
    gameHour: 14,
  };
}
