/**
 * legacy-asset-loader.ts — resilient loader for hand-drawn Kwame + environment.
 *
 * - Retries on 429 / transient network errors
 * - Never aborts the whole world if one optional animation is missing
 * - Prefers atlas sheets when available; still supports individual frames
 */

import "pixi.js/unsafe-eval";

import { Assets, Texture, Rectangle } from "pixi.js";
import type { LegacyFullAnimState } from "@/lib/legacy-combat-fsm";
import type { LegacyFacing } from "@/lib/legacy-animation-fsm";

export interface CharacterFrameSet {
  [key: string]: Texture[]; // key = `${animState}:${facing}`
}

export interface CharacterFrameManifestEntry {
  animState: LegacyFullAnimState;
  facing: LegacyFacing;
  /** Ordered frame file paths, relative to baseUrl */
  frameFiles: string[];
}

export interface CharacterManifest {
  characterId: string;
  baseUrl: string;
  frames: CharacterFrameManifestEntry[];
}

const frameSetCache = new Map<string, CharacterFrameSet>();

const MAX_RETRIES = 3;
const RETRY_BASE_MS = 400;

async function loadWithRetry(urls: string[]): Promise<Texture[]> {
  const results: Texture[] = [];

  for (const url of urls) {
    let lastError: unknown = null;
    let loaded: Texture | null = null;

    for (let attempt = 0; attempt < MAX_RETRIES; attempt++) {
      try {
        loaded = await Assets.load<Texture>(url);
        break;
      } catch (err) {
        lastError = err;
        const msg = String(err);
        const isRateLimit = msg.includes("429") || msg.includes("Too Many Requests");
        const isMissing = msg.includes("404") || msg.includes("Failed to fetch");

        if (isMissing) {
          // Don't retry hard 404s forever
          console.warn(`[legacy-asset-loader] missing frame (skip): ${url}`);
          break;
        }

        // Back-off on 429 / transient errors
        const delay = RETRY_BASE_MS * Math.pow(2, attempt) + Math.random() * 200;
        console.warn(`[legacy-asset-loader] retry ${attempt + 1}/${MAX_RETRIES} for ${url} (${isRateLimit ? "429" : "error"})`);
        await new Promise((r) => setTimeout(r, delay));
      }
    }

    results.push(loaded ?? Texture.WHITE);
  }

  return results;
}

export async function loadCharacterFrameSet(manifest: CharacterManifest): Promise<CharacterFrameSet> {
  const cached = frameSetCache.get(manifest.characterId);
  if (cached) return cached;

  const result: CharacterFrameSet = {};

  // Load each animation group with resilience — one missing group must not kill the world
  for (const entry of manifest.frames) {
    const urls = entry.frameFiles.map((f) => manifest.baseUrl + f);
    try {
      const ordered = await loadWithRetry(urls);
      // Keep only real textures; if everything failed, leave empty so resolveFrames can fall back
      const valid = ordered.filter((t) => t && t !== Texture.WHITE);
      if (valid.length > 0) {
        result[frameKey(entry.animState, entry.facing)] = valid;
      } else {
        console.warn(
          `[legacy-asset-loader] no valid frames for ${entry.animState}:${entry.facing} — will fall back at runtime`
        );
      }
    } catch (err) {
      console.warn(`[legacy-asset-loader] failed group ${entry.animState}:${entry.facing}`, err);
    }
  }

  // Guarantee at least an idle:down so the character always appears
  if (!result["idle:down"]?.length) {
    result["idle:down"] = [Texture.WHITE];
  }

  frameSetCache.set(manifest.characterId, result);
  return result;
}

export function frameKey(animState: LegacyFullAnimState, facing: LegacyFacing): string {
  return `${animState}:${facing}`;
}

export function resolveFrames(
  frameSet: CharacterFrameSet,
  animState: LegacyFullAnimState,
  facing: LegacyFacing
): { frames: Texture[]; isFallback: boolean } {
  const exact = frameSet[frameKey(animState, facing)];
  if (exact?.length) return { frames: exact, isFallback: false };

  const downFallback = frameSet[frameKey(animState, "down")];
  if (downFallback?.length) return { frames: downFallback, isFallback: true };

  const idleFallback = frameSet[frameKey("idle", facing)] ?? frameSet[frameKey("idle", "down")];
  return { frames: idleFallback?.length ? idleFallback : [Texture.WHITE], isFallback: true };
}

// ─── Environment ─────────────────────────────────────────────────────────────

export interface EnvironmentManifestEntry {
  assetId: string;
  file: string;
}

export async function loadEnvironmentTextures(
  baseUrl: string,
  entries: EnvironmentManifestEntry[]
): Promise<Map<string, Texture>> {
  const byAssetId = new Map<string, Texture>();
  const urls = entries.map((e) => baseUrl + e.file);

  // Load in small batches to reduce 429 pressure
  const BATCH = 6;
  for (let i = 0; i < urls.length; i += BATCH) {
    const batchUrls = urls.slice(i, i + BATCH);
    const batchEntries = entries.slice(i, i + BATCH);

    try {
      const loaded = await loadWithRetry(batchUrls);
      batchEntries.forEach((e, idx) => {
        byAssetId.set(e.assetId, loaded[idx] ?? Texture.WHITE);
      });
    } catch (err) {
      console.warn("[legacy-asset-loader] environment batch failed", err);
      batchEntries.forEach((e) => byAssetId.set(e.assetId, Texture.WHITE));
    }

    // Small pause between batches under rate-limit pressure
    if (i + BATCH < urls.length) {
      await new Promise((r) => setTimeout(r, 120));
    }
  }

  return byAssetId;
}

/** Slice a uniform grid atlas into frames (left→right, top→bottom) */
export function sliceSheet(
  sheet: Texture,
  frameWidth: number,
  frameHeight: number,
  frameCount: number
): Texture[] {
  const frames: Texture[] = [];
  const cols = Math.floor(sheet.width / frameWidth);
  for (let i = 0; i < frameCount; i++) {
    const col = i % cols;
    const row = Math.floor(i / cols);
    frames.push(
      new Texture({
        source: sheet.source,
        frame: new Rectangle(col * frameWidth, row * frameHeight, frameWidth, frameHeight),
      })
    );
  }
  return frames;
}

import { Assets, Texture, Rectangle } from "pixi.js";

/** Slice a uniform grid atlas into ordered frames (left→right, top→bottom). */
export function sliceSheet(
  sheet: Texture,
  frameWidth: number,
  frameHeight: number,
  frameCount: number
): Texture[] {
  const frames: Texture[] = [];
  const cols = Math.max(1, Math.floor(sheet.width / frameWidth));

  for (let i = 0; i < frameCount; i++) {
    const col = i % cols;
    const row = Math.floor(i / cols);
    const x = col * frameWidth;
    const y = row * frameHeight;

    // Guard against reading past the sheet edge
    if (x + frameWidth > sheet.width + 1 || y + frameHeight > sheet.height + 1) {
      break;
    }

    frames.push(
      new Texture({
        source: sheet.source,
        frame: new Rectangle(x, y, frameWidth, frameHeight),
      })
    );
  }

  return frames.length > 0 ? frames : [Texture.WHITE];
}

export interface SheetManifestEntry {
  animState: LegacyFullAnimState;
  facing: LegacyFacing;
  /** Path relative to baseUrl, e.g. "sheets/pick-up-sheet.png" */
  sheetFile: string;
  frameWidth: number;
  frameHeight: number;
  frameCount: number;
  /** Optional: start frame index if the sheet contains multiple animations */
  startFrame?: number;
}

export interface SheetBasedCharacterManifest {
  characterId: string;
  baseUrl: string; // e.g. "/legacy-character-assets/kwame-mensah/"
  sheets: SheetManifestEntry[];
}

/**
 * Load character animations from whole atlas sheets (the format your ZIPs actually contain).
 * One HTTP request per animation instead of 6–32.
 */
export async function loadCharacterFrameSetFromSheets(
  manifest: SheetBasedCharacterManifest
): Promise<CharacterFrameSet> {
  const cached = frameSetCache.get(manifest.characterId);
  if (cached) return cached;

  const result: CharacterFrameSet = {};

  for (const entry of manifest.sheets) {
    const url = manifest.baseUrl + entry.sheetFile;
    try {
      const sheet = await Assets.load<Texture>(url);
      const allFrames = sliceSheet(
        sheet,
        entry.frameWidth,
        entry.frameHeight,
        entry.frameCount + (entry.startFrame ?? 0)
      );

      const start = entry.startFrame ?? 0;
      const frames = allFrames.slice(start, start + entry.frameCount);

      if (frames.length > 0) {
        result[frameKey(entry.animState, entry.facing)] = frames;
      } else {
        console.warn(`[sheet-loader] empty slice for ${entry.animState}:${entry.facing}`);
      }
    } catch (err) {
      console.warn(`[sheet-loader] failed to load sheet ${url}`, err);
      // continue — other animations still load
    }
  }

  // Guarantee a visible character
  if (!result["idle:down"]?.length) {
    result["idle:down"] = [Texture.WHITE];
  }

  frameSetCache.set(manifest.characterId, result);
  return result;
}

function ensureDir(dir) {
  if (!fs.existsSync(dir)) {
    if (dryRun) console.log(`[dry-run] mkdir -p ${dir}`);
    else fs.mkdirSync(dir, { recursive: true });
  }
}

function unzipToTemp(zipPath, label) {
  if (!fs.existsSync(zipPath)) {
    console.error(`✖ Missing ZIP: ${zipPath}`);
    process.exit(1);
  }
  const tmp = path.join("/tmp", `niakofa-${label}-${Date.now()}`);
  ensureDir(tmp);
  console.log(`→ Unzipping ${path.basename(zipPath)} …`);
  execSync(`unzip -q -o "${zipPath}" -d "${tmp}"`, { stdio: "inherit" });
  return tmp;
}

function findFiles(dir, filterFn) {
  const results = [];
  function walk(d) {
    for (const name of fs.readdirSync(d)) {
      if (name === "__MACOSX" || name === ".DS_Store") continue;
      const full = path.join(d, name);
      const st = fs.statSync(full);
      if (st.isDirectory()) walk(full);
      else if (filterFn(name)) results.push(full);
    }
  }
  walk(dir);
  return results;
}

    if (!targetName) {
      skipped++;
      continue;
    }
    const dest = path.join(outDir, targetName);
    if (dryRun) {
      console.log(`[dry-run] ${label}: ${base}  →  ${targetName}`);
    } else {
      fs.copyFileSync(file, dest);
      console.log(`  ✓ ${label}: ${targetName}`);
    }
    copied++;
  }

  console.log(`  ${label}: ${copied} copied, ${skipped} ignored (not in map)`);
  return copied;
}

// ── Main ───────────────────────────────────────────────────────────────────
console.log("Niakofa hand-drawn asset installer");

const kwameTmp = unzipToTemp(KWAME_ZIP, "kwame");

const kwamePngs = findFiles(kwameTmp, (n) => n.toLowerCase().endsWith(".png"));

console.log("\nDone.");
console.log(`  Character sheets → ${OUT_KWAME}`);
console.log(`  Environment sheets → ${OUT_ENV}`);
console.log(`  Total installed: ${c1 + c2}`);

if (!dryRun) {
  console.log("\nNext:");
  console.log("  1. Confirm files exist under public/…/sheets/");
  console.log("  2. Use KWAME_SHEET_MANIFEST + loadCharacterFrameSetFromSheets");

How to run it
Option A – ZIPs already in the repo attached_assets/ folder
Bash

# from repo root
node scripts/install-hand-drawn-assets.mjs


Option B – ZIPs are somewhere else (e.g. Downloads or the attachments path you used)
Bash

node scripts/install-hand-drawn-assets.mjs --src "/home/workdir/attachments"
# or
node scripts/install-hand-drawn-assets.mjs --src "/path/to/your/zips"


Dry-run first (recommended)
Bash

node scripts/install-hand-drawn-assets.mjs --src "/home/workdir/attachments" --dry-run



What the script does
1. Unzips both archives into temporary folders.
2. Finds every .png (ignores __MACOSX and .DS_Store).
3. Renames only the known atlas sheets to the clean names expected by KWAME_SHEET_MANIFEST.
4. Copies them into:
    * artifacts/pay-it-forward/public/legacy-character-assets/kwame-mensah/sheets/
    * artifacts/pay-it-forward/public/legacy-environment-assets/sheets/
5. Leaves concept / ChatGPT reference images out of the runtime path (or copies them with ref- names if you want them available).

After the script finishes
1. Confirm the sheets exist:Bash   ls artifacts/pay-it-forward/public/legacy-character-assets/kwame-mensah/sheets/
2. ls artifacts/pay-it-forward/public/legacy-environment-assets/sheets/      
3. Make sure LegacyGameCanvas uses:TypeScript   loadCharacterFrameSetFromSheets(KWAME_SHEET_MANIFEST)      
4. Deploy and hard-refresh the demo.
The 429 errors from individual frame URLs should stop, and the living world will load the real hand-drawn atlas sheets instead.



If any sheet name in the ZIP differs slightly from the map (typo, extra space, etc.), the dry-run output will show which files were ignored so you can add them to KWAME_MAP / ENV_MAP.
